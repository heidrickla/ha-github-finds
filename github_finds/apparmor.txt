#include <tunables/global>

profile github_finds flags=(attach_disconnected,mediate_deleted) {
  #include <abstractions/base>
  #include <abstractions/nameservice>
  #include <abstractions/ssl_certs>

  # Files: the rule set the Home Assistant example profiles use for an
  # s6-overlay based image, so init and the server can run.
  file,
  signal (send) set=(kill,term,int,hup,cont),
  signal (receive) set=(kill,term,int,hup,cont),

  /init ix,
  /bin/** ix,
  /usr/bin/** ix,
  /run/{s6,s6-rc*,service}/** ix,
  /package/** ix,
  /command/** ix,
  /etc/services.d/** rwix,
  /etc/cont-init.d/** rwix,
  /etc/cont-finish.d/** rwix,
  /run/{,**} rwk,
  /dev/tty rw,
  /tmp/** rwk,

  # Options and discovery state; the links file in the add-on's config folder.
  /data/** rw,
  /config/** rw,

  # The server, a static Rust binary, and curl for Docker's health check.
  /usr/bin/github-finds rmix,
  /usr/bin/curl rmix,

  # The page on its port and through ingress; GitHub's search and the model router outward.
  network inet stream,
  network inet6 stream,
  network inet dgram,
  network inet6 dgram,
}
